How the directives are defined

// 1. Attribute / Element directive — adds .highlighted class
app.directive('highlight', function() {
  return {
    restrict: 'AE',          // A = attribute, E = element
    link: function(scope, element) {
      element.addClass('highlighted');
    }
  };
});

// 2. Attribute directive that reads its own value for color
app.directive('highlightColor', function() {
  return {
    restrict: 'A',
    link: function(scope, element, attrs) {
      attrs.$observe('highlightColor', function(color) {
        element.css('color', color || 'blue');
        element.css('font-weight', 'bold');
      });
    }
  };
});

/* CSS applied by directive 1 */
.highlighted { color: red; font-weight: bold; }
